function getDbKey_(sheetName) {
  if (sheetName === 'TempatBerangkat') return 'tempatBerangkat';
  return sheetName.toLowerCase();
}


function serializeSheetValue_(value, header) {
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) {
    const key = String(header || '').toLowerCase();
    if (key === 'createdat') {
      return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
    }
    if (key.indexOf('tanggal') !== -1 || key.indexOf('tgl') !== -1) {
      return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd');
    }
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  }
  return value;
}

function loadAllData() {
  const ss = SpreadsheetApp.openById(MAIN_SPREADSHEET_ID);
  const sheets = ['Pegawai', 'Jabatan', 'Rekening', 'Shs', 'Lokasi', 'TempatBerangkat', 'Pertimbangan', 'Sppd'];
  const db = {};
  
  sheets.forEach(sheetName => {
    const sheet = ss.getSheetByName(sheetName);
    if (sheet) {
      const data = sheet.getDataRange().getValues();
      if (data.length > 1) {
        const headers = data[0].map(h => String(h).trim());
        const rows = [];
        for (let i = 1; i < data.length; i++) {
          let obj = {};
          for (let j = 0; j < headers.length; j++) {
            let val = serializeSheetValue_(data[i][j], headers[j]);
            // Jika data berupa JSON String (seperti array pengikut), parsing kembali
            try {
               if (typeof val === 'string' && (val.startsWith('{') || val.startsWith('['))) {
                 val = JSON.parse(val);
               }
            } catch(e){}
            obj[headers[j]] = val;
          }
          rows.push(obj);
        }
        db[getDbKey_(sheetName)] = rows;
      } else {
        db[getDbKey_(sheetName)] = [];
      }
    } else {
       db[getDbKey_(sheetName)] = [];
    }
  });
  
  // Memuat Pengaturan dari Sheet 'Settings'
  let settings = null;
  const setSheet = ss.getSheetByName('Settings');
  if(setSheet && setSheet.getLastRow() > 0) {
     const setData = setSheet.getDataRange().getValues();
     try {
         // Menggabungkan kembali potongan-potongan string dari baris A1, A2, A3, dst
         const fullString = setData.map(row => row[0]).join('');
         settings = JSON.parse(fullString);
     } catch(e) {}
  }

  return { db: db, settings: settings, absensiStatus: {} };
}

  return loadAbsensiStatus_(tanggalBerangkat, pegawaiIds || []);

function loadAbsensiStatus_(tanggalBerangkat, pegawaiIds) {
  const statusMap = {};
  const targetDateKey = normalizeDateKey_(tanggalBerangkat);
  if (!targetDateKey) return statusMap;

  const allowedIds = {};
  (pegawaiIds || []).forEach(id => {
    const key = String(id || '').trim();
    if (key) allowedIds[key] = true;
  });
  const filterById = Object.keys(allowedIds).length > 0;

  try {
    const ssAbsensi = SpreadsheetApp.openById(ABSENSI_SPREADSHEET_ID);
    const sheet = ssAbsensi.getSheetByName(ABSENSI_SHEET_NAME);
    if (!sheet || sheet.getLastRow() < 2) return statusMap;

    const values = sheet.getDataRange().getValues();
    // Kolom B = tanggal absensi, Kolom E = ID_QR, Kolom K = Keterangan/status kehadiran, Kolom L = keterangan TL.
    // Dibaca dari bawah agar data absensi terbaru per pegawai pada tanggal berangkat menjadi prioritas.
    for (let i = values.length - 1; i >= 1; i--) {
      const row = values[i];
      const rowDateKey = normalizeDateKey_(row[1]);
      if (rowDateKey !== targetDateKey) continue;

      const idQr = String(row[4] || '').trim();
      if (!idQr || statusMap[idQr] || (filterById && !allowedIds[idQr])) continue;

      const status = String(row[10] || '').trim().toUpperCase();
      const detailTl = String(row[11] || '').trim();
      statusMap[idQr] = {
        id: idQr,
        tanggal: targetDateKey,
        status: status,
        detailTl: detailTl,
        rowNumber: i + 1,
        updatedAt: new Date().toISOString()
      };
    }
  } catch (error) {
    console.error('Gagal membaca spreadsheet absensi: ' + error.message);
  }
  return statusMap;
}

function normalizeDateKey_(value) {
  if (!value) return '';
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }

  const text = String(value).trim();
  if (!text) return '';
  const isoMatch = text.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (isoMatch) return isoMatch[1] + '-' + isoMatch[2] + '-' + isoMatch[3];

  const slashMatch = text.match(/^(\d{1,2})[\/-](\d{1,2})[\/-](\d{4})$/);
  if (slashMatch) {
    return slashMatch[3] + '-' + ('0' + slashMatch[2]).slice(-2) + '-' + ('0' + slashMatch[1]).slice(-2);
  }

  const parsed = new Date(text);
  if (!isNaN(parsed.getTime())) return Utilities.formatDate(parsed, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  return text;
}

function ensureDocumentTemplates(settingsPayload) {
  const settings = settingsPayload || {};
  const templateIds = settings.templateIds || {};
  const templates = getDefaultTemplateDefinitions_();

  Object.keys(templates).forEach(key => {
    if (!templateIds[key]) {
      const doc = DocumentApp.create('Template e-SPPD ' + templates[key].name);
      const body = doc.getBody();
      body.clear();
      templates[key].content.split('\n').forEach(line => appendSafeParagraph_(body, line));
      doc.saveAndClose();
      templateIds[key] = doc.getId();
    }
  });

  settings.templateIds = templateIds;
  return {
    templateIds: templateIds,
    editUrls: Object.keys(templateIds).reduce((acc, key) => {
      acc[key] = 'https://docs.google.com/document/d/' + templateIds[key] + '/edit';
      return acc;
    }, {})
  };
}

function generateSppdPdfFromTemplates(sppdData, dbPayload, settingsPayload, printOptions) {
  const settings = settingsPayload || {};
  const ensured = ensureDocumentTemplates(settings);
  const data = buildTemplateData_(sppdData || {}, dbPayload || {}, settings);
  const output = DocumentApp.create('Dokumen SPPD - ' + (data.NOMOR_SPD || data.NOMOR_SPT || new Date().getTime()));
  applyDocumentLayout_(output, printOptions || {});
  const body = output.getBody();
  body.clear();

  const kopLogoBlob = createImageBlobFromDataUrl_(settings.kopLogo, 'kop-logo.png');
  ['spt', 'spd', 'kwitansi', 'laporan'].forEach((key, index) => {
    const templateId = ensured.templateIds[key];
    const templateDoc = DocumentApp.openById(templateId);
    if (index > 0) body.appendPageBreak();
    const sectionStartIndex = body.getNumChildren();
    appendBodyElements_(body, templateDoc.getBody());
    insertKopLogo_(body, sectionStartIndex, kopLogoBlob);
  });

  insertSptEmployeeTable_(body, data.__PEGAWAI_SPT || []);
  insertSpdEmployeeTable_(body, data.__PEGAWAI_SPD || []);
  fillBodyPlaceholders_(body, data);
  output.saveAndClose();
  const pdfBlob = DriveApp.getFileById(output.getId()).getAs(MimeType.PDF).setName(output.getName() + '.pdf');
  const pdfFile = DriveApp.createFile(pdfBlob);
  return {
    status: 'success',
    templateIds: ensured.templateIds,
    docUrl: 'https://docs.google.com/document/d/' + output.getId() + '/edit',
    pdfUrl: pdfFile.getUrl()
  };
}


function appendBodyElements_(targetBody, sourceBody) {
  const count = sourceBody.getNumChildren();
  for (let i = 0; i < count; i++) {
    const child = sourceBody.getChild(i);
    const copy = child.copy();
    const type = child.getType();
    if (type === DocumentApp.ElementType.PARAGRAPH) {
      appendSafeParagraph_(targetBody, copy.asParagraph());
    } else if (type === DocumentApp.ElementType.TABLE) {
      targetBody.appendTable(copy.asTable());
    } else if (type === DocumentApp.ElementType.LIST_ITEM) {
      appendSafeListItem_(targetBody, copy.asListItem());
    } else if (type === DocumentApp.ElementType.HORIZONTAL_RULE) {
      targetBody.appendHorizontalRule();
    } else if (type === DocumentApp.ElementType.PAGE_BREAK) {
      targetBody.appendPageBreak();
    } else if (child.getText && child.getText()) {
      appendSafeParagraph_(targetBody, child.getText());
    }
  }
}

function normalizeDocumentText_(value) {
  const text = String(value === undefined || value === null ? '' : value);
  return text.length ? text : '\u00A0';
}

function appendSafeParagraph_(body, paragraphOrText) {
  if (paragraphOrText && paragraphOrText.getType && paragraphOrText.getType() === DocumentApp.ElementType.PARAGRAPH) {
    const paragraph = paragraphOrText.asParagraph();
    return paragraph.getText() ? body.appendParagraph(paragraph) : body.appendParagraph('\u00A0');
  }
  return body.appendParagraph(normalizeDocumentText_(paragraphOrText));
}

function appendSafeListItem_(body, listItem) {
  if (listItem && listItem.getText && listItem.getText()) return body.appendListItem(listItem);
  return body.appendListItem('\u00A0');
}

function fillBodyPlaceholders_(body, data) {
  Object.keys(data || {}).forEach(key => {
    if (key.indexOf('__') === 0) return;
    body.replaceText('\\{\\{' + key + '\\}\\}', normalizeDocumentText_(data[key]));
  });
}

function createImageBlobFromDataUrl_(dataUrl, name) {
  const text = String(dataUrl || '').trim();
  const match = text.match(/^data:(image\/[a-zA-Z0-9.+-]+);base64,(.+)$/);
  if (!match) return null;
  try {
    return Utilities.newBlob(Utilities.base64Decode(match[2]), match[1], name || 'image.png');
  } catch (error) {
    console.error('Gagal membaca logo kop: ' + error.message);
    return null;
  }
}

function insertKopLogo_(body, childIndex, logoBlob) {
  if (!logoBlob) return;
  try {
    const image = body.insertImage(Math.max(0, childIndex), logoBlob.copyBlob());
    image.setWidth(58).setHeight(58);
  } catch (error) {
    console.error('Gagal menyisipkan logo kop: ' + error.message);
  }
}

function insertSptEmployeeTable_(body, people) {
  const placeholder = '{{TABEL_PEGAWAI_SPT}}';
  const rows = buildSptEmployeeRows_(people || []);

  // Placeholder bisa diletakkan sebagai paragraf biasa atau di dalam sel tabel template.
  // Jika berada di dalam sel tabel, tabel pegawai disisipkan sebagai nested table transparan
  // agar menyatu dengan layout tabel Google Docs yang sudah diedit pengguna.
  for (let i = body.getNumChildren() - 1; i >= 0; i--) {
    const child = body.getChild(i);
    const type = child.getType();
    if (type === DocumentApp.ElementType.PARAGRAPH) {
      const paragraph = child.asParagraph();
      if (paragraph.getText().indexOf(placeholder) === -1) continue;
      const table = body.insertTable(i + 1, rows);
      styleSptEmployeeTable_(table);
      body.removeChild(paragraph);
    } else if (type === DocumentApp.ElementType.TABLE) {
      replaceDynamicTablePlaceholderInTable_(child.asTable(), placeholder, rows, styleSptEmployeeTable_);
    }
  }
}

function replaceDynamicTablePlaceholderInTable_(table, placeholder, rows, styler) {
  for (let r = 0; r < table.getNumRows(); r++) {
    const row = table.getRow(r);
    for (let c = 0; c < row.getNumCells(); c++) {
      replaceDynamicTablePlaceholderInCell_(row.getCell(c), placeholder, rows, styler);
    }
  }
}

function replaceDynamicTablePlaceholderInCell_(cell, placeholder, rows, styler) {
  for (let i = 0; i < cell.getNumChildren(); i++) {
    const child = cell.getChild(i);
    const type = child.getType();
    if (type === DocumentApp.ElementType.PARAGRAPH) {
      const paragraph = child.asParagraph();
      if (paragraph.getText().indexOf(placeholder) === -1) continue;
      const remainingText = paragraph.getText().split(placeholder).join('').trim();
      paragraph.setText(remainingText || '\u00A0');
      const nestedTable = cell.appendTable(rows);
      styler(nestedTable);
    } else if (type === DocumentApp.ElementType.TABLE) {
      replaceDynamicTablePlaceholderInTable_(child.asTable(), placeholder, rows, styler);
    }
  }
}

function buildSpdEmployeeRows_(people) {
  const rows = [[
    'No',
    'Nama / NIP',
    'Pangkat / Gol.',
    'Jabatan',
    'Nomor SPD'
  ]];
  (people.length ? people : [{}]).forEach((p, index) => {
    const pangkatGol = [p.pangkat, p.golongan].filter(Boolean).join(' / ');
    rows.push([
      safeCellText_(String(index + 1) + '.'),
      safeCellText_([p.nama, p.nip ? 'NIP. ' + p.nip : ''].filter(Boolean).join('\n')),
      safeCellText_(pangkatGol || '-'),
      safeCellText_(p.jabatan || '-'),
      safeCellText_(p.nomorSpd || '-')
    ]);
  });
  return rows;
}

function styleSpdEmployeeTable_(table) {
  table.setBorderWidth(0.5);
  for (let r = 0; r < table.getNumRows(); r++) {
    const row = table.getRow(r);
    row.getCell(0).setWidth(18);
    row.getCell(1).setWidth(142);
    row.getCell(2).setWidth(82);
    row.getCell(3).setWidth(118);
    row.getCell(4).setWidth(128);
    for (let c = 0; c < row.getNumCells(); c++) {
      const cell = row.getCell(c);
      cell.setPaddingTop(0).setPaddingBottom(0).setPaddingLeft(1).setPaddingRight(1);
      cell.editAsText().setFontSize(8);
      if (r === 0) cell.editAsText().setBold(true);
      for (let i = 0; i < cell.getNumChildren(); i++) {
        const child = cell.getChild(i);
        if (child.getType() === DocumentApp.ElementType.PARAGRAPH) {
          child.asParagraph().setSpacingBefore(0).setSpacingAfter(0).setLineSpacing(1);
        }
      }
    }
  }
}

function buildSptEmployeeRows_(people) {
  const rows = [];
  (people.length ? people : [{}]).forEach((p, index) => {
    const pangkatGol = [p.pangkat, p.golongan].filter(Boolean).join(' / ');
    rows.push([safeCellText_(String(index + 1) + '.'), 'Nama', ':', safeCellText_(p.nama || '-')]);
    rows.push([safeCellText_(' '), 'Pangkat/Golongan', ':', safeCellText_(pangkatGol || '-')]);
    rows.push([safeCellText_(' '), 'Jabatan', ':', safeCellText_(p.jabatan || '-')]);
  });
  return rows;
}

function safeCellText_(value) {
  const text = String(value === undefined || value === null ? '' : value);
  return text.trim() ? text : '\u00A0';
}

function applyDocumentLayout_(doc, options) {
  const body = doc.getBody();
  const mmToPt = mm => Number(mm || 0) * 72 / 25.4;
  const sizes = {
    f4: [215.9, 330.2],
    a4: [210, 297],
    legal: [215.9, 355.6]
  };
  const selected = sizes[String(options.paperSize || 'f4').toLowerCase()] || sizes.f4;
  const isLandscape = options.orientation === 'landscape';
  body.setPageWidth(mmToPt(isLandscape ? selected[1] : selected[0]));
  body.setPageHeight(mmToPt(isLandscape ? selected[0] : selected[1]));
  body.setMarginTop(mmToPt(options.marginTop || 15));
  body.setMarginRight(mmToPt(options.marginRight || 15));
  body.setMarginBottom(mmToPt(options.marginBottom || 15));
  body.setMarginLeft(mmToPt(options.marginLeft || 15));
}

    TEMPAT_BERANGKAT: berangkat.tempat || 'UPTD Puskesmas Tanjung Puri',
    TUJUAN: tujuan.tujuan || tujuan.lokasi || tujuan.nama_lokasi || tujuan.tujuan_lokasi || '-',
    PEGAWAI_UTAMA: utama.nama || '-',
    DAFTAR_PEGAWAI: people.map((p, i) => (i + 1) + '. ' + (p.nama || '-') + ' - ' + (p.jabatan || '') + ' / NIP. ' + (p.nip || '')).join('\n'),
    DAFTAR_PEGAWAI_SPT: daftarPegawaiSpt,
    __PEGAWAI_SPT: people,
    __PEGAWAI_SPD: peopleSpd,
    SUMBER_DANA: sumberDana || '-',
    REKENING: [rekening.kode_rek_kegiatan + ' ' + rekening.nama_kegiatan, rekening.kode_rek_sub_kegiatan + ' ' + rekening.nama_sub_Kegiatan, rekening.kode_rek_objek_belanja + ' ' + rekening.nama_objek_belanja].filter(Boolean).join('\n'),
    UANG_HARIAN: formatNumber_(uangHarian),
    TRANSPORT: formatNumber_(transport),
    TOTAL_BIAYA: formatNumber_(Number(sppd.nilai_sppd || total)),
    LAPORAN_HASIL: sppd.laporanHasil || 'Belum ada uraian laporan kegiatan yang diinput.'
  };
}


function normalizeBaseSpdNumber_(baseNumber) {
  return String(baseNumber || '').trim().replace(/-\d+$/, '');
}

function buildPersonSpdNumber_(baseNumber, index) {
  const text = normalizeBaseSpdNumber_(baseNumber);
  const increment = Number(index || 0);
  if (!text) return String(increment + 1).padStart(3, '0') + '/SPD';
  const match = text.match(/^(\d+)(\/SPD\/.+)$/i);
  if (!match) return text;
  return String(Number(match[1]) + increment).padStart(match[1].length, '0') + match[2];
}

function formatJabatanGolongan_(jabatan, golongan) {
  const jab = String(jabatan || '').trim();
  const gol = String(golongan || '').trim();
  if (jab && gol && jab.indexOf(gol) === -1) return jab + '/(' + gol + ')';
  return jab || gol || '';
}

function formatDisplayDate_(value) {
  const key = normalizeDateKey_(value);
  const match = key.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!match) return value ? String(value) : '';
  const months = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'];
  const monthIndex = Number(match[2]) - 1;
  if (monthIndex < 0 || monthIndex > 11) return value ? String(value) : '';
  return match[3] + ' ' + months[monthIndex] + ' ' + match[1];
}

function fillTemplate_(text, data) {
  let output = text || '';
  Object.keys(data).forEach(key => {
    output = output.split('{{' + key + '}}').join(data[key]);
  });
  return output;
}

function findById_(rows, id) {
  return (rows || []).find(row => String(row.id) === String(id));
}

function formatNumber_(num) {
  return Number(num || 0).toLocaleString('id-ID');
}

function normalizeDateTimeForSheet_(value) {
  if (!value) return '';
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  }
  const text = String(value).trim();
  if (!text || text === '-') return '';
  const normalizedText = text.replace('T', ' ').replace(/\.\d{3}Z?$/, '').replace(/Z$/, '');
  const match = normalizedText.match(/^(\d{4})-(\d{2})-(\d{2})(?:\s+(\d{2}):(\d{2})(?::(\d{2}))?)?/);
  if (match) return match[1] + '-' + match[2] + '-' + match[3] + ' ' + (match[4] || '00') + ':' + (match[5] || '00') + ':' + (match[6] || '00');
  const parsed = new Date(text);
  if (!isNaN(parsed.getTime())) return Utilities.formatDate(parsed, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  return text;
}

// ==========================================
function saveDatabase(dbPayload, settingsPayload) {
  const ss = SpreadsheetApp.openById(MAIN_SPREADSHEET_ID);
  const lock = LockService.getScriptLock();
  
  // Mencegah bentrok jika ada banyak user menyimpan bersamaan
  lock.tryLock(10000); 
  
  try {
    // 1. Simpan Data Master & Transaksi (Tabel DB)
    for (let key in dbPayload) {
      const sheetName = key === 'tempatBerangkat' ? 'TempatBerangkat' : key.charAt(0).toUpperCase() + key.slice(1); // Kapitalisasi nama sheet
      let sheet = ss.getSheetByName(sheetName);
      
      // Jika sheet belum ada, buat otomatis
      if (!sheet) {
        sheet = ss.insertSheet(sheetName);
      }
      
      sheet.clear(); // Bersihkan data lama
      
      const dataArray = dbPayload[key];
      const defaultHeaders = {
        pegawai: ['id', 'nip', 'nama', 'pangkat', 'golongan', 'jabatan'],
        jabatan: ['id', 'jabatan'],
        rekening: ['id', 'nama_kegiatan', 'kode_rek_kegiatan', 'nama_sub_Kegiatan', 'kode_rek_sub_kegiatan', 'sumberDana', 'kode_rek_objek_belanja', 'nama_objek_belanja', 'pagu_dana'],
        shs: ['id', 'kriteria', 'uangHarian', 'penginapan'],
        lokasi: ['id', 'tujuan', 'jarak', 'transport'],
        tempatBerangkat: ['id', 'tempat', 'keterangan'],
        pertimbangan: ['id', 'dasar_pertimbangan'],
        sppd: ['id', 'nomor', 'nomorSpt', 'nomorSpd', 'rekeningId', 'rekeningKegiatanKey', 'rekeningSubKegiatanKey', 'maksud', 'tempatBerangkatId', 'lokasiId', 'tglBerangkat', 'tglKembali', 'lamaHari', 'pegawaiUtama', 'pengikut', 'tanggal_spt', 'tanggal_spd', 'tanggal_kwitansi', 'laporanHasil', 'createdAt', 'dasar_pertimbangan', 'nilai_sppd']
      };
      const baseHeaders = defaultHeaders[key] || (dataArray && dataArray.length > 0 ? Object.keys(dataArray[0]) : []);
      const extraHeaders = [];
      (dataArray || []).forEach(item => Object.keys(item).forEach(h => { if (baseHeaders.indexOf(h) === -1 && extraHeaders.indexOf(h) === -1) extraHeaders.push(h); }));
      const headers = baseHeaders.concat(extraHeaders);
      if (headers.length > 0) {
        const rows = [headers];
        
        (dataArray || []).forEach(item => {
          const row = headers.map(h => {
             let val = item[h];
             if (h === 'createdAt') return normalizeDateTimeForSheet_(val);
             // Ubah Array/Object menjadi string agar bisa masuk ke dalam sel Excel
             if (typeof val === 'object' && val !== null) return JSON.stringify(val);
             return val === undefined ? '' : val;
          });
          rows.push(row);
        });
        
        // Tulis semua baris baru ke dalam sheet sekaligus agar cepat
        sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
      }
    }

    // 2. Simpan Pengaturan & Logo dengan metode Chunking (Pemotongan)
    // Untuk menghindari limit 50.000 karakter per cell di Google Sheets
    if(settingsPayload) {
      let setSheet = ss.getSheetByName('Settings');
      if(!setSheet) setSheet = ss.insertSheet('Settings');
      setSheet.clear();
      
      const jsonString = JSON.stringify(settingsPayload);
      const chunkSize = 45000; // Batas aman di bawah 50.000 karakter
      const chunks = [];
      
      for (let i = 0; i < jsonString.length; i += chunkSize) {
        chunks.push([jsonString.substring(i, i + chunkSize)]);
      }
      
      // Menyimpan tiap potongan ke baris yang berurutan di Kolom A
      if (chunks.length > 0) {
        setSheet.getRange(1, 1, chunks.length, 1).setValues(chunks);
      }
    }
    
    return { status: 'success', message: 'Tersimpan di Google Sheets' };
  } catch (error) {
    return { status: 'error', message: error.message };
  } finally {
    lock.releaseLock();
  }
}
