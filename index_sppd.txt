                normalizeDateValue(value) {
                    if (!value) return '';
                    if (value instanceof Date && !Number.isNaN(value.getTime())) return value.toISOString().slice(0, 10);
                    const text = String(value).trim();
                    if (!text || text === '-') return '';
                    const isoMatch = text.match(/^(\d{4})-(\d{2})-(\d{2})/);
                    if (isoMatch) return `${isoMatch[1]}-${isoMatch[2]}-${isoMatch[3]}`;
                    const localMatch = text.match(/^(\d{1,2})[\/-](\d{1,2})[\/-](\d{4})$/);
                    if (localMatch) return `${localMatch[3]}-${String(localMatch[2]).padStart(2, '0')}-${String(localMatch[1]).padStart(2, '0')}`;
                    const parsed = new Date(text);
                    return Number.isNaN(parsed.getTime()) ? text : parsed.toISOString().slice(0, 10);
                },
                formatDisplayDate(value) {
                    const normalized = this.normalizeDateValue(value);
                    const match = normalized.match(/^(\d{4})-(\d{2})-(\d{2})$/);
                    if (!match) return value || '';
                    const months = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'];
                    const monthIndex = Number(match[2]) - 1;
                    if (monthIndex < 0 || monthIndex > 11) return value || '';
                    return `${match[3]} ${months[monthIndex]} ${match[1]}`;
                },
                normalizeDateTimeValue(value) {
                    if (!value) return '';
                    const pad = number => String(number).padStart(2, '0');
                    const fromDate = date => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())} ${pad(date.getHours())}:${pad(date.getMinutes())}:${pad(date.getSeconds())}`;
                    if (value instanceof Date && !Number.isNaN(value.getTime())) return fromDate(value);
                    const text = String(value).trim();
                    if (!text || text === '-') return '';
                    const normalizedText = text.replace('T', ' ').replace(/\.\d{3}Z?$/, '').replace(/Z$/, '');
                    const match = normalizedText.match(/^(\d{4})-(\d{2})-(\d{2})(?:\s+(\d{2}):(\d{2})(?::(\d{2}))?)?/);
                    if (match) return `${match[1]}-${match[2]}-${match[3]} ${match[4] || '00'}:${match[5] || '00'}:${match[6] || '00'}`;
                    const parsed = new Date(text);
                    return Number.isNaN(parsed.getTime()) ? text : fromDate(parsed);
                },
                formatDateTimeDisplay(value) {
                    const normalized = this.normalizeDateTimeValue(value);
                    const match = normalized.match(/^(\d{4})-(\d{2})-(\d{2})\s+(\d{2}):(\d{2})/);
                    if (!match) return normalized || '-';
                    return `${match[3]}-${match[2]}-${match[1]} ${match[4]}:${match[5]}`;
                },

                    const selectedPegawaiIds = [this.formSppd.pegawaiUtama, ...this.formSppd.pengikut.map(p => p.pegawaiId)];
                    const blockedPegawai = selectedPegawaiIds.map(id => ({ id, nama: this.getPegawaiName(id), info: this.getAbsensiInfo(id) })).filter(item => !item.info.available);
                    if(blockedPegawai.length > 0) {
                        Swal.fire({ icon: 'warning', title: 'Pegawai tidak bisa berangkat', html: `<b>${blockedPegawai.length}</b> pegawai diblokir, <b>${Math.max(selectedPegawaiIds.length - blockedPegawai.length, 0)}</b> pegawai bisa digunakan.` }); return;
                    }
                    if(!this.formSppd.rekeningId) {
                        Swal.fire('Rekening belum lengkap', 'Pilih kegiatan, sub kegiatan, dan objek belanja secara berurutan.', 'warning'); return;
                    }
                    if(!this.formSppd.tempatBerangkatId) {
                        Swal.fire('Tempat berangkat belum dipilih', 'Pilih tempat berangkat dari master Tempat Berangkat.', 'warning'); return;
                    }
                    if(!this.formSppd.tanggal_spt || !this.formSppd.tanggal_spd || !this.formSppd.tanggal_kwitansi) {
                        Swal.fire('Tanggal surat belum lengkap', 'Isi tanggal SPT, tanggal SPD, dan tanggal kwitansi.', 'warning'); return;
                    }
                    if(this.formSppd.lamaHari <= 0) {
                        Swal.fire('Error', 'Tanggal kembali harus setelah atau sama dengan tanggal berangkat', 'error'); return;
                    }
                    this.refreshNomorSurat();
                    const newSppd = { ...this.formSppd, pengikut: this.formSppd.pengikut.map(({ search, ...rest }) => rest), id: Date.now().toString(), createdAt: this.normalizeDateTimeValue(new Date()) };
                    this.db.sppd.push(newSppd);
                    this.saveDB();
                    Swal.fire('Berhasil', 'Data SPPD tersimpan. Silahkan klik Cetak Dokumen.', 'success');
                    this.resetFormSppd();
                },

                },

                        this.settings.kopLogo = e.target.result;

                    reader.readAsDataURL(file);

                // GENERATE PDF PDF LOGIC (jsPDF + Permendagri)

                generatePDF(sppdData) {
                    if (this.isGAS) {
                        this.isSyncing = true;
                        google.script.run
                            .withSuccessHandler((res) => {
                                this.isSyncing = false;
                                if (res.templateIds) this.settings.templateIds = res.templateIds;
                                if (res.pdfUrl) window.open(res.pdfUrl, '_blank');
                                Swal.fire({ icon: 'success', title: 'PDF Siap', text: 'Dokumen dibuat dari template Google Docs.' });
                            })
                            .withFailureHandler((err) => {
                                this.isSyncing = false;
                                Swal.fire({ icon: 'error', title: 'Gagal Generate PDF', text: err.message });
                            })
                            .generateSppdPdfFromTemplates(sppdData, this.db, this.settings);
                        return;
                    }
                    try {
                        const paperDimensions = {
                            'f4': [215.9, 330.2],
                            'legal': [215.9, 355.6],
                            'a4': [210, 297]
                        };
                        const dim = paperDimensions[this.settings.paperSize] || paperDimensions['f4'];
                        const doc = new jspdf.jsPDF({ orientation: 'p', unit: 'mm', format: dim });
                        
                        // Helpers
                        const margin = this.settings.marginTop;
                        const centerX = dim[0] / 2;
                        const pageWidth = dim[0];
                        let y = margin;
                        
                        const utama = this.db.pegawai.find(p => p.id === sppdData.pegawaiUtama);
                        const lokasi = this.db.lokasi.find(l => String(l.id) === String(sppdData.lokasiId));
                        const tempatBerangkat = this.getTempatBerangkatName(sppdData.tempatBerangkatId);
                        const rekening = this.getRekeningInfo(sppdData);
                        const tanggalSpt = this.formatDisplayDate(sppdData.tanggal_spt || sppdData.tglBerangkat);
                        const tanggalSpd = this.formatDisplayDate(sppdData.tanggal_spd || sppdData.tglBerangkat);
                        const tanggalKwitansi = this.formatDisplayDate(sppdData.tanggal_kwitansi || sppdData.tglBerangkat);
                        const tglBerangkatDisplay = this.formatDisplayDate(sppdData.tglBerangkat);
                        const tglKembaliDisplay = this.formatDisplayDate(sppdData.tglKembali);
                        
                        const drawKopSurat = () => {
                            let currentY = y;
                            if (this.settings.kopLogo) {
                                try { doc.addImage(this.settings.kopLogo, 15, currentY, 22, 26); } catch (e) { }
                            }

                            doc.setFont("times", "bold");
                            doc.setFontSize(14);
                            doc.text(this.settings.kopInstansi || '', centerX, currentY, { align: 'center' });
                            doc.setFontSize(16);
                            doc.text(this.settings.kopDinas || '', centerX, currentY+6, { align: 'center' });
                            doc.setFontSize(14);
                            doc.text(this.settings.kopUptd || '', centerX, currentY+12, { align: 'center' });
                            doc.setFont("times", "normal");
                            doc.setFontSize(10);
                            
                            let alamatLines = doc.splitTextToSize(this.settings.kopAlamat || '', pageWidth - 40);
                            doc.text(alamatLines, centerX, currentY+17, { align: 'center' });
                            
                            y += 20 + (alamatLines.length * 4);
                            doc.setLineWidth(0.5); doc.line(15, y, pageWidth-15, y);
                            doc.setLineWidth(1.0); doc.line(15, y+1, pageWidth-15, y+1);
                            y += 8;
                        };

                        const printTtd = (yPos, tanggalSurat = sppdData.tglBerangkat) => {
                            doc.setFont("times", "normal");
                            doc.text("Dikeluarkan di : Sintang", pageWidth - 70, yPos);
                            doc.text(`Pada tanggal   : ${tanggalSurat || tglBerangkatDisplay}`, pageWidth - 70, yPos + 5);
                            doc.setFont("times", "bold");
                            doc.text("KEPALA UPTD PUSKESMAS", pageWidth - 70, yPos + 12);
                            doc.text("TANJUNG PURI", pageWidth - 70, yPos + 17);
                            doc.text(this.settings.kepalaNama, pageWidth - 70, yPos + 40);
                            doc.setFont("times", "normal");
                            doc.text(`NIP. ${this.settings.kepalaNip}`, pageWidth - 70, yPos + 45);
                        };

                        // ----------------- DOKUMEN 1: SPT -----------------
                        drawKopSurat();
                        doc.setFont("times", "bold"); doc.setFontSize(12);
                        doc.text("SURAT PERINTAH TUGAS", centerX, y, { align: 'center' });
                        doc.line(centerX - 30, y+1, centerX + 30, y+1);
                        doc.setFont("times", "normal");
                        doc.text(`Nomor: ${sppdData.nomorSpt || sppdData.nomor}`, centerX, y+6, { align: 'center' });
                        y += 15;

                        doc.setFontSize(11);
                        doc.text("Dasar", 20, y); doc.text(":", 45, y);
                        let dasarText = doc.splitTextToSize("DPA SKPD UPTD Puskesmas Tanjung Puri Tahun Anggaran 2026", pageWidth - 55);
                        doc.text(dasarText, 50, y);
                        y += 10;
                        
                        doc.setFont("times", "bold");
                        doc.text("MEMERINTAHKAN :", centerX, y, { align: 'center' });
                        y += 8; doc.setFont("times", "normal");

                        let people = [{...utama, ket: 'Utama'}];
                        sppdData.pengikut.forEach(png => {
                            let pData = this.db.pegawai.find(p => p.id === png.pegawaiId);
                            if(pData) people.push({...pData, ket: 'Pengikut'});
                        });

                        doc.text("Kepada", 20, y); doc.text(":", 45, y);
                        people.forEach((person, idx) => {
                            doc.text(`${idx+1}.`, 50, y);
                            doc.text(`Nama`, 55, y);     doc.text(`: ${person.nama}`, 75, y); y+=5;
                            doc.text(`NIP`, 55, y);      doc.text(`: ${person.nip}`, 75, y); y+=5;
                            doc.text(`Pangkat/Gol`, 55, y); doc.text(`: ${person.pangkat} / ${person.golongan}`, 75, y); y+=5;
                            doc.text(`Jabatan`, 55, y);  doc.text(`: ${person.jabatan}`, 75, y); y+=8;
                        });

                        doc.text("Untuk", 20, y); doc.text(":", 45, y);
                        let maksudLines = doc.splitTextToSize(sppdData.maksud, pageWidth - 55);
                        doc.text(maksudLines, 50, y);
                        y += (maksudLines.length * 5) + 5;

                        doc.text("Waktu", 20, y); doc.text(":", 45, y);
                        doc.text(`${sppdData.lamaHari} Hari (${tglBerangkatDisplay} s/d ${tglKembaliDisplay})`, 50, y);
                        y += 10;
                        doc.text("Berangkat", 20, y); doc.text(":", 45, y);
                        doc.text(tempatBerangkat, 50, y);
                        y += 7;
                        doc.text("Tujuan", 20, y); doc.text(":", 45, y);
                        doc.text(lokasi ? lokasi.tujuan : '-', 50, y);
                        y += 20;

                        printTtd(y, tanggalSpt);

                        // ----------------- DOKUMEN 2: SPD -----------------
                        doc.addPage(); y = margin; drawKopSurat();
                        doc.setFont("times", "bold");
                        doc.text("SURAT PERJALANAN DINAS (SPD)", centerX, y, { align: 'center' });
                        doc.setFont("times", "normal");
                        doc.text(`Nomor: ${sppdData.nomorSpd || sppdData.nomor}`, centerX, y+6, { align: 'center' });
                        y += 14;

                        const spdBody = [
                            ["1", "Pejabat berwenang yang memberi perintah", "Kepala UPTD Puskesmas Tanjung Puri"],
                            ["2", "Nama Pegawai yang diperintah", utama ? utama.nama : ''],
                            ["3", "a. Pangkat dan Golongan\nb. Jabatan", `${utama?utama.pangkat:''} / ${utama?utama.golongan:''}\n${utama?utama.jabatan:''}`],
                            ["4", "Maksud Perjalanan Dinas", sppdData.maksud],
                            ["5", "Alat angkutan yang dipergunakan", "Kendaraan Umum / Dinas"],
                            ["6", "a. Tempat berangkat\nb. Tempat tujuan", `${tempatBerangkat}\n${lokasi?lokasi.tujuan:''}`],
                            ["7", "a. Lama perjalanan dinas\nb. Tanggal berangkat\nc. Tanggal kembali", `${sppdData.lamaHari} Hari\n${tglBerangkatDisplay}\n${tglKembaliDisplay}`],
                            ["8", "Pembebanan Anggaran\na. Instansi\nb. Rekening", `Puskesmas Tanjung Puri\n${this.getRekeningLabel(rekening)}`]
                        ];

                        doc.autoTable({
                            startY: y, body: spdBody, theme: 'grid',
                            styles: { font: 'times', fontSize: 10, cellPadding: 2, textColor: [0,0,0] },
                            columnStyles: { 0: { cellWidth: 10 }, 1: { cellWidth: 60 } }
                        });

                        y = doc.lastAutoTable.finalY + 15;
                        printTtd(y, tanggalSpd);

                        // ----------------- DOKUMEN 3: KWITANSI -----------------
                        doc.addPage(); y = margin; drawKopSurat();
                        doc.setFont("times", "bold");
                        doc.text("RINCIAN BIAYA PERJALANAN DINAS", centerX, y, { align: 'center' });
                        y += 15;
                        
                        let tarifUH = 100000;
                        if(this.db.shs.length > 0) tarifUH = this.db.shs[0].uangHarian;
                        let transport = lokasi ? lokasi.transport : 50000;
                        let totalOrang = people.length;
                        let totalBiayaUH = tarifUH * sppdData.lamaHari * totalOrang;
                        let totalTransport = transport * totalOrang;
                        let grandTotal = totalBiayaUH + totalTransport;

                        const kuitansiBody = [
                            ["1", `Uang Harian (${sppdData.lamaHari} Hari x ${totalOrang} Orang)`, `Rp ${this.formatNumber(tarifUH)}`, `Rp ${this.formatNumber(totalBiayaUH)}`],
                            ["2", `Biaya Transportasi PP (${totalOrang} Orang)`, `Rp ${this.formatNumber(transport)}`, `Rp ${this.formatNumber(totalTransport)}`],
                            ["", "TOTAL BIAYA", "", `Rp ${this.formatNumber(grandTotal)}`]
                        ];

                        doc.autoTable({
                            startY: y, head: [['No', 'Rincian Biaya', 'Satuan', 'Jumlah']],
                            body: kuitansiBody, theme: 'grid',
                            styles: { font: 'times', fontSize: 10, textColor: [0,0,0] },
                            headStyles: { fillColor: [240,240,240], textColor: [0,0,0] }
                        });

                        y = doc.lastAutoTable.finalY + 20; doc.setFont("times", "normal");
                        doc.text("Telah Menerima Jumlah Uang Tersebut,", 20, y);
                        doc.text("Yang Menerima", 20, y+5);
                        doc.text(utama ? utama.nama : '', 20, y+30);
                        doc.text(`NIP. ${utama ? utama.nip : ''}`, 20, y+35);
                        
                        doc.text(`Lunas Dibayar Tanggal : ${tanggalKwitansi || '.........................'}`, pageWidth - 80, y);
                        doc.text("Bendahara Pengeluaran,", pageWidth - 80, y+5);
                        doc.text("(.........................................)", pageWidth - 80, y+30);

                        // ----------------- DOKUMEN 4: LAPORAN -----------------
                        doc.addPage(); y = margin; drawKopSurat();
                        doc.setFont("times", "bold");
                        doc.text("LAPORAN HASIL PERJALANAN DINAS", centerX, y, { align: 'center' });
                        doc.setLineWidth(0.5); doc.line(centerX - 40, y+1, centerX + 40, y+1);
                        y += 15; doc.setFontSize(11); doc.setFont("times", "normal");
                        
                        const textDasar = doc.splitTextToSize("DPA SKPD UPTD Puskesmas Tanjung Puri", pageWidth - 70);
                        doc.text("1.", 20, y); doc.text("Dasar", 25, y); doc.text(":", 65, y); doc.text(textDasar, 70, y);
                        y += (textDasar.length * 5) + 2;

                        const textMaksud = doc.splitTextToSize(sppdData.maksud, pageWidth - 70);
                        doc.text("2.", 20, y); doc.text("Maksud Tujuan", 25, y); doc.text(":", 65, y); doc.text(textMaksud, 70, y);
                        y += (textMaksud.length * 5) + 2;

                        doc.text("3.", 20, y); doc.text("Waktu Pelaksanaan", 25, y); doc.text(":", 65, y); 
                        doc.text(`${sppdData.lamaHari} Hari (${tglBerangkatDisplay} s/d ${tglKembaliDisplay})`, 70, y);
                        y += 7;

                        doc.text("4.", 20, y); doc.text("Nama Petugas", 25, y); doc.text(":", 65, y); 
                        people.forEach((p, i) => { doc.text(`${i+1}. ${p.nama}`, 70, y); y += 5; });
                        y += 2;

                        const textDaerah = doc.splitTextToSize(lokasi ? lokasi.tujuan : '-', pageWidth - 70);
                        doc.text("5.", 20, y); doc.text("Daerah Tujuan", 25, y); doc.text(":", 65, y); doc.text(textDaerah, 70, y);
                        y += (textDaerah.length * 5) + 2;

                        doc.text("6.", 20, y); doc.text("Hasil Kegiatan", 25, y); doc.text(":", 65, y); 
                        let textLaporanRaw = sppdData.laporanHasil || "Belum ada uraian laporan kegiatan yang diinput.";
                        let textLaporanLines = doc.splitTextToSize(textLaporanRaw, pageWidth - 70);
                        
                        for(let i=0; i < textLaporanLines.length; i++) {
                            if (y > (dim[1] - margin - 30)) { doc.addPage(); y = margin + 10; }
                            doc.text(textLaporanLines[i], 70, y); y += 5;
                        }

                        y += 15;
                        if (y > (dim[1] - margin - 40)) { doc.addPage(); y = margin + 10; }
                        doc.text("Sintang, ..............................", pageWidth - 80, y);
                        doc.text("Pelapor,", pageWidth - 80, y+5);
                        doc.text(utama ? utama.nama : '', pageWidth - 80, y+30);
                        doc.text(`NIP. ${utama ? utama.nip : ''}`, pageWidth - 80, y+35);

                        const blob = doc.output('blob');
                        const url = URL.createObjectURL(blob);
                        window.open(url, '_blank');
                        
                        Swal.fire('Berhasil', 'Dokumen PDF (SPT, SPD, Kwitansi, Laporan) berhasil di-generate.', 'success');

                    } catch (error) {
                        console.error(error);
                        Swal.fire('Error', 'Gagal memproses PDF: ' + error.message, 'error');
                    }
                }
            }
        }).mount('#app');
